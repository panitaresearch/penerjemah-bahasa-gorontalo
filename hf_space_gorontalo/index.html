🔤 Jembatan Konversi Ortografi Pateda (Ejaan 1987 ↔ Ejaan Baku 2001)

Kamus Indonesia–Gorontalo (1987) menggunakan vokal ganda (aa, ii, uu, ee, oo), tanda tanya glotal (?), dan tanpa luncuran (tio, boito, mai), sedangkan Kamus Gorontalo–Indonesia (2001) menggunakan titik dua (:), tanpa tanda glotal, dan luncuran eksplisit (tiyo, boyito, mayi).

Langkah 2 — Menjalankan Model 1: panitagorontalo/nllb-200-gorontalo-bi-600M (v6 — Ringan, Cepat, BLEU 40.59)

import re, torch
from transformers import AutoTokenizer, AutoModelForSeq2SeqLM

MODEL_ID = "panitagorontalo/nllb-200-gorontalo-bi-600M"
device = "cuda" if torch.cuda.is_available() else "cpu"
tokenizer = AutoTokenizer.from_pretrained(MODEL_ID)
model = AutoModelForSeq2SeqLM.from_pretrained(MODEL_ID).to(device)

def translate_nllb(text, src_lang="ind_Latn", tgt_lang="gor_Latn"):
    src_id = tokenizer.convert_tokens_to_ids(src_lang)
    tgt_id = tokenizer.convert_tokens_to_ids(tgt_lang)
    sents = [s.strip() for s in re.split(r"(?<=[.!?])\s+", text.strip()) if s.strip()]
    out_parts = []
    for sent in sents:
        sub_ids = tokenizer(sent, add_special_tokens=False)["input_ids"]
        inp = torch.tensor([[src_id] + sub_ids + [tokenizer.eos_token_id]], device=device)
        gen = model.generate(inp, forced_bos_token_id=tgt_id, max_new_tokens=160, num_beams=4, no_repeat_ngram_size=4)
        pred = tokenizer.decode(gen[0], skip_special_tokens=True).strip()
        if len(sents) > 1 and pred and not pred.endswith((".", "!", "?")):
            pred += sent[-1] if sent[-1] in ".!?" else "."
        out_parts.append(pred)
    return " ".join(out_parts)

print(translate_nllb("Saya pergi ke pasar kemarin.", "ind_Latn", "gor_Latn"))

Langkah 3 — Menjalankan Model 2: panitagorontalo/qwen2.5-7b-gorontalo-lora (v6 — 4-bit untuk T4 GPU Gratis Colab)

import torch
from peft import PeftModel
from transformers import AutoTokenizer, AutoModelForCausalLM, BitsAndBytesConfig

BASE_LLM = "Qwen/Qwen2.5-7B-Instruct"
LORA_ADAPTER = "panitagorontalo/qwen2.5-7b-gorontalo-lora"
bnb_config = BitsAndBytesConfig(load_in_4bit=True, bnb_4bit_quant_type="nf4", bnb_4bit_compute_dtype=torch.float16)

qwen_tok = AutoTokenizer.from_pretrained(LORA_ADAPTER)
base_model = AutoModelForCausalLM.from_pretrained(BASE_LLM, quantization_config=bnb_config, device_map="auto")
qwen_model = PeftModel.from_pretrained(base_model, LORA_ADAPTER)
